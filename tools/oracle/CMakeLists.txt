cmake_minimum_required(VERSION 3.20)
project(moonclip_oracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED CLIPPER2_SOURCE OR NOT DEFINED JSON_HEADER_DIR)
  message(FATAL_ERROR "Run scripts/build_oracle.sh to verify source identities first")
endif()
add_library(clipper2_reference STATIC "${CLIPPER2_SOURCE}/CPP/Clipper2Lib/src/clipper.engine.cpp" "${CLIPPER2_SOURCE}/CPP/Clipper2Lib/src/clipper.offset.cpp" "${CLIPPER2_SOURCE}/CPP/Clipper2Lib/src/clipper.rectclip.cpp")
target_compile_definitions(clipper2_reference PUBLIC CLIPPER2_HI_PRECISION=0)
target_include_directories(clipper2_reference PUBLIC "${CLIPPER2_SOURCE}/CPP/Clipper2Lib/include")
add_executable(moonclip-oracle adapter.cpp)
target_include_directories(moonclip-oracle PRIVATE "${JSON_HEADER_DIR}")
target_link_libraries(moonclip-oracle PRIVATE clipper2_reference)
if(MSVC)
  target_compile_options(moonclip-oracle PRIVATE /W4 /WX)
else()
  target_compile_options(moonclip-oracle PRIVATE -Wall -Wextra -Werror)
endif()
